def match_route(path, method)
    @routes.each do |route|
        if route[:path] == path && route[:method] == method
            return [route, {}]
        elsif (params = route[:path].match(path)) && route[:method] == method
            return [route, params.named_captures(symbolize_names: true)]
        end
    end
    return nil
end
