def self.build(data, session = nil)
    if data.start_with? "GET"
        return GetRequest.new(data)
    elsif data.start_with? "POST"
        return PostRequest.new(data, session)
    elsif data.start_with? "HEAD"
        return HeadRequest.new(data)
    else
        raise MethodError
    end
end
